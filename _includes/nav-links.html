{%- comment -%} Primary navigation links from `_data/navigation.yml`, rendered once for the desktop bar and once for the mobile panel so the two can never drift. Usage: {% include nav-links.html variant="desktop" %} | {% include nav-links.html variant="mobile" %} Params: variant — "desktop" (pill row) or "mobile" (stacked rows). Items with `module:` are skipped when that module is off in `_data/site.yml`. `style: button` renders the primary call to action. The current page gets aria-current="page". Local assigns are prefixed `nl_`. {%- endcomment -%} {%- assign nl_variant = include.variant | default: 'desktop' -%} {%- assign nl_cfg = site.data.site -%} {%- assign nl_current = page.url | default: '/' -%} {%- for nl_item in site.data.navigation -%} {%- if nl_item.module -%} {%- assign nl_enabled = nl_cfg.modules[nl_item.module] -%} {%- unless nl_enabled -%}{%- continue -%}{%- endunless -%} {%- endif -%} {%- assign nl_href = nl_item.url | default: '/' -%} {%- assign nl_active = false -%} {%- if nl_current == nl_href -%} {%- assign nl_active = true -%} {%- elsif nl_href != '/' and nl_current contains nl_href -%} {%- assign nl_active = true -%} {%- endif -%} {%- if nl_item.style == 'button' -%} {%- if nl_variant == 'mobile' -%} {{ nl_item.label }} {%- else -%} {{ nl_item.label }} {%- endif -%} {%- elsif nl_variant == 'mobile' -%} {{ nl_item.label }} {%- else -%} {{ nl_item.label }} {%- endif -%} {%- endfor -%}
